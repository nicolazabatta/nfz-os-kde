# Container tools
podman
buildah
skopeo
ostree

# Time sync (keeps the clock right, e.g. after the VM was suspended)
chrony

# English locale data (LANG=en_US.UTF-8 in /etc/locale.conf); the minimal base
# only has C/POSIX, so Qt and other programs warned "Failed to set locale"
glibc-langpack-en

# Command-line tools
usbutils            # lsusb
curl
openssl             # create and inspect certificates (openssl-libs is already a dependency)
git
ncurses
zsh

# Archiving and compression (command line; also used by Krusader and Ark)
tar
gzip
bzip2
xz
zstd
zip
unzip
7zip                # 7z, and extracts RAR

# Disks and filesystems (used by Partition Manager, Plasma and Krusader)
udisks2             # mounting USB sticks and external drives from Plasma/Krusader
smartmontools       # smartctl: SMART health data
btrfs-progs
ntfs-3g             # read/write NTFS (Windows drives)
ntfsprogs           # format/resize NTFS
exfatprogs          # exFAT (SD cards, large USB sticks)
cryptsetup          # LUKS encrypted drives
lvm2

# Needed by Private Internet Access (PIA's app is installed per machine from its .run file)
iptables-nft        # iptables (kill switch, firewall rules)
iproute             # ip (routes, interfaces)
libnsl              # network library used by PIA's daemon
xterm               # PIA 3.8+ requires it (its GUI opens commands in xterm)
